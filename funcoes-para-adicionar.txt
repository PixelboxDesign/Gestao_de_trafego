// ÔöÇÔöÇ Drag and Drop para Carrossel ÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇÔöÇ
let draggedElement = null;

function handleDragStart(e) {
  draggedElement = e.target;
  e.target.style.opacity = '0.4';
  e.dataTransfer.effectAllowed = 'move';
}

function handleDragOver(e) {
  if (e.preventDefault) {
    e.preventDefault();
  }
  e.dataTransfer.dropEffect = 'move';
  
  // Visual feedback
  const target = e.target.closest('.carrossel-item');
  if (target && target !== draggedElement) {
    target.style.borderColor = '#3b82f6';
    target.style.borderWidth = '2px';
  }
  
  return false;
}

function handleDrop(e) {
  if (e.stopPropagation) {
    e.stopPropagation();
  }
  
  const target = e.target.closest('.carrossel-item');
  if (!target || !draggedElement || target === draggedElement) {
    return false;
  }
  
  // Remove visual feedback
  target.style.borderColor = '';
  target.style.borderWidth = '';
  
  // Pega ├¡ndices
  const fromIndex = parseInt(draggedElement.getAttribute('data-index'));
  const toIndex = parseInt(target.getAttribute('data-index'));
  
  // Reordena array de imagens
  const imagens = [...(kitAtual.imagens_carrossel || [])];
  const [movedImage] = imagens.splice(fromIndex, 1);
  imagens.splice(toIndex, 0, movedImage);
  
  // Atualiza kitAtual
  kitAtual.imagens_carrossel = imagens;
  
  // Re-renderiza
  renderCarrossel(imagens);
  
  // Salva automaticamente a nova ordem
  salvarOrdemCarrossel(imagens);
  
  return false;
}

function handleDragEnd(e) {
  e.target.style.opacity = '1';
  
  // Remove todos os feedbacks visuais
  document.querySelectorAll('.carrossel-item').forEach(item => {
    item.style.borderColor = '';
    item.style.borderWidth = '';
  });
  
  draggedElement = null;
}

async function salvarOrdemCarrossel(imagens) {
  if (!kitAtual) return;

function handleThumbSelect(input) {
  const file = input.files?.[0];
  if (!file) {
    arquivoThumb = null;
    document.getElementById('file-name-thumb').textContent = '';
    return;
  }

  // Valida├º├úo de tipo
  const validTypes = ['image/jpeg', 'image/jpg', 'image/png', 'image/webp'];
  if (!validTypes.includes(file.type)) {
    alert('ÔÜá´©Å Formato inv├ílido. Use apenas: JPG, PNG ou WebP');
    input.value = '';
    arquivoThumb = null;
    document.getElementById('file-name-thumb').textContent = '';
    return;
  }

  // Valida├º├úo de tamanho (5MB)
  const maxSize = 5 * 1024 * 1024;
  if (file.size > maxSize) {
    alert('ÔÜá´©Å Arquivo muito grande. Tamanho m├íximo: 5MB');
    input.value = '';
    arquivoThumb = null;
    document.getElementById('file-name-thumb').textContent = '';
    return;
  }

  arquivoThumb = file;
  document.getElementById('file-name-thumb').textContent = `Ô£ô ${file.name} (${(file.size / 1024).toFixed(1)} KB)`;
}

async function uploadCarrossel() {
  if (!arquivoCarrossel || !kitAtual) return;

  const formData = new FormData();
  formData.append('imagem', arquivoCarrossel);

  try {
    const res = await fetch(`/api/catalogo/upload-carrossel/${encodeURIComponent(kitAtual.marca)}/${encodeURIComponent(kitAtual.nome)}`, {
      method: 'POST',
      body: formData
    });

    if (!res.ok) {
      const err = await res.text();
      throw new Error(err);
    }

    // Recarrega dados do kit
    await carregarCatalogo();
    kitAtual = kitsData.find(k => k.nome === kitAtual.nome);
    renderCarrossel(kitAtual.imagens_carrossel || []);
    
    // Limpa input
    document.getElementById('modal-kit-file-carrossel').value = '';
    arquivoCarrossel = null;

    showToast('Ô£ô Imagem adicionada ao carrossel', 'success');
  } catch (err) {
    console.error('Erro ao upload carrossel:', err);
    alert('ÔØî Erro ao adicionar imagem: ' + err.message);
  }
}

async function salvarKit() {
  if (!kitAtual) return;

  const btn = document.getElementById('modal-save-btn');
  const novoNome = document.getElementById('modal-kit-nome-edit').value.trim();
  const preco = document.getElementById('modal-kit-preco').value.trim();
  const descricao = document.getElementById('modal-kit-descricao').value.trim();
  const skuKit = document.getElementById('modal-kit-sku').value.trim();

  btn.disabled = true;
  btn.textContent = 'Salvando...';

  try {
    // 1. Atualizar dados do kit via API v2
    const marca = kitAtual.marca || 'Alphahall';
    
    // Converte pre├ºo para n├║mero
    const precoLimpo = preco.replace('R$ ', '').replace(',', '.');
    const precoNumero = parseFloat(precoLimpo) || 0;
    
    const res = await fetch(`/api/catalogo/v2/kit/${encodeURIComponent(marca)}/${encodeURIComponent(kitAtual.nome)}`, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        preco: precoNumero,  // Envia como n├║mero, n├úo string
        codigo_sku: skuKit,
        descricao: descricao
      })
    });

    if (!res.ok) {
      const err = await res.text();
      throw new Error(err || 'Erro ao salvar');
    }

    // 2. Upload de thumbnail (se selecionou)
    if (arquivoThumb) {
      const formData = new FormData();
      formData.append('imagem', arquivoThumb);

      const uploadRes = await fetch(`/api/catalogo/upload-thumb/${encodeURIComponent(marca)}/${encodeURIComponent(kitAtual.nome)}?tipo=kit`, {
        method: 'POST',
        body: formData
      });

      if (!uploadRes.ok) {
        const err = await uploadRes.text();
        throw new Error('Erro no upload da thumb: ' + err);
      }
    }

    // Atualiza dados locais
    kitAtual.preco = preco.replace('R$ ', '').replace(',', '.');
    kitAtual.codigo_sku = skuKit;
    kitAtual.descricao = descricao;

    // Mostra sucesso
    showToast('Ô£ô Kit salvo com sucesso', 'success');
    document.getElementById('modal-save-success').style.display = 'block';
    setTimeout(() => {
      document.getElementById('modal-save-success').style.display = 'none';
    }, 2000);

    // Limpa arquivo thumb selecionado
    arquivoThumb = null;
    document.getElementById('file-name-thumb').textContent = '';
    document.getElementById('modal-kit-file-thumb').value = '';

    // Recarrega cat├ílogo
    await carregarCatalogo();

  } catch (e) {
    console.error('Erro ao salvar kit:', e);
    alert('Erro ao salvar: ' + e.message);
  } finally {
    btn.disabled = false;
    btn.textContent = '­ƒÆ¥ Salvar';
  }
}

async function carregarProdutos() {
  try {
    const data = await api(`/api/catalogo/v2/produtos-individuais`);
    produtosData = data || [];
    console.log('[DEBUG] Produtos carregados:', produtosData.length, '- Exemplo:', produtosData[0]);
    renderProdutos(produtosData);
  } catch (e) {
    console.error('[Produtos] Erro ao carregar:', e);
    document.getElementById('produtos-grid').innerHTML = `
      <div class="empty-state" style="grid-column:1/-1">
        <span>ÔÜá´©Å</span>
        <p>Erro ao carregar produtos</p>
        <p style="font-size:11px;color:var(--text2)">Verifique se o servidor est├í rodando</p>
      </div>`;
  }
}

function renderProdutos(lista) {
  const grid = document.getElementById('produtos-grid');
  const count = document.getElementById('produtos-count');
  
  count.textContent = `${lista.length} produto${lista.length !== 1 ? 's' : ''}`;

  console.log('[DEBUG] renderProdutos - total:', lista.length);

  if (lista.length === 0) {
    grid.innerHTML = `
      <div class="empty-state" style="grid-column:1/-1">
        <span>­ƒº┤</span>
        <p>Nenhum produto no cat├ílogo</p>
      </div>`;
    return;
  }

  grid.innerHTML = lista.map(produto => {
    const temThumb = produto.caminho_thumb_bd || produto.tem_thumb;
    const marca = produto.marca || 'Alphahall'; // Fallback para Alphahall se marca estiver vazia
    const thumbUrl = produto.caminho_thumb_bd 
      ? `/api/catalogo/files/${encodeURIComponent(produto.caminho_thumb_bd)}` 
      : produto.tem_thumb 
        ? `/api/catalogo/imagem/${encodeURIComponent(marca)}/${encodeURIComponent(produto.nome)}/thumb.${produto.thumb_ext}?tipo=produto` 
        : null;
    
    console.log('[DEBUG] Renderizando produto:', produto.nome, '- marca:', marca, '- data-produto-marca:', escAttr(marca));
    
    return `
    <div class="kit-card" data-produto-nome="${escAttr(produto.nome)}" data-produto-marca="${escAttr(marca)}">
      <div class="kit-img-area">
        ${temThumb
          ? `<img 
              loading="lazy" 
              data-src="${thumbUrl}" 
              alt="${esc(produto.nome)}" 
              class="lazy-img"
              style="background:var(--bg3)"
            />`
          : `<span class="kit-img-placeholder">­ƒº┤</span>`
